cmake_minimum_required(VERSION 3.16)
project(snia_cs_api VERSION 1.1 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
add_library(snia-cs-api SHARED
    src/cs_discovery.cpp src/cs_access.cpp src/cs_afdm_management.cpp
    src/cs_storage_ios.cpp src/cs_csf_scheduling.cpp src/cs_batch_scheduling.cpp
    src/cs_event_management.cpp src/cs_management.cpp src/cs_library_management.cpp
    src/cs_handle_registry.cpp src/cs_request_execution.cpp)
target_include_directories(snia-cs-api PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
target_link_libraries(snia-cs-api PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
target_link_options(snia-cs-api PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
set_property(TARGET snia-cs-api APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
add_subdirectory(plugins/femu)
install(TARGETS snia-cs-api LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/cs.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
foreach(example api-smoke live-api)
    if(example STREQUAL "api-smoke")
        set(example_source examples/api-smoke.c)
    else()
        set(example_source tests/plugin_tests/live-api.c)
    endif()
    add_executable(${example} ${example_source})
    target_compile_options(${example} PRIVATE -UNDEBUG)
    target_link_libraries(${example} PRIVATE snia-cs-api Threads::Threads)
    target_include_directories(${example} PRIVATE plugins/femu/include)
endforeach()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
