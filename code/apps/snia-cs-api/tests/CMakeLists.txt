# A separate facade library loads mock plugins from its own adjacent folder.
# Tests still exercise dlopen and the production facade/core code unchanged.
get_target_property(API_SOURCES snia-cs-api SOURCES)
list(TRANSFORM API_SOURCES PREPEND "${PROJECT_SOURCE_DIR}/")
add_library(test-api-library SHARED ${API_SOURCES})
target_include_directories(test-api-library PRIVATE "${PROJECT_SOURCE_DIR}/include" "${PROJECT_SOURCE_DIR}/src/core" "${PROJECT_SOURCE_DIR}/src/interface")
target_link_libraries(test-api-library PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(test-api-library PROPERTIES OUTPUT_NAME snia-cs-api
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-runtime")
femu_plugin(mock-femu plugin_tests/mock-nvme.c "${CMAKE_CURRENT_BINARY_DIR}/test-runtime/snia-cs-api")
target_link_options(mock-femu PRIVATE "LINKER:--wrap=glob,--wrap=globfree")
target_compile_definitions(mock-femu PRIVATE CS_TEST_PROGRAM="${PROJECT_SOURCE_DIR}/../../csd-programs/build/snia-cs-scale.so")
add_executable(test-api plugin_tests/test-api.c)
target_link_libraries(test-api PRIVATE test-api-library Threads::Threads ${CMAKE_DL_LIBS})
target_include_directories(test-api PRIVATE "${PROJECT_SOURCE_DIR}/include" "${FEMU_DIR}/include")
target_compile_definitions(test-api PRIVATE CS_MOCK_PLUGIN="$<TARGET_FILE:mock-femu>")
add_dependencies(test-api mock-femu)
add_test(NAME femu-api COMMAND test-api)
set_tests_properties(femu-api PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
