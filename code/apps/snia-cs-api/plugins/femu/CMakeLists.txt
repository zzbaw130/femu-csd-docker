set(FEMU_DIR "${CMAKE_CURRENT_SOURCE_DIR}" PARENT_SCOPE)
set(FEMU_SOURCES)
foreach(source plugin_entry discovery catalog profile memory request validation management spill)
    list(APPEND FEMU_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/${source}.c")
endforeach()
set(FEMU_SOURCES "${FEMU_SOURCES}" PARENT_SCOPE)
function(femu_plugin target transport destination)
    add_library(${target} MODULE ${FEMU_SOURCES} ${transport})
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/include" "${PROJECT_SOURCE_DIR}/src" "${FEMU_DIR}" "${FEMU_DIR}/include")
    target_link_libraries(${target} PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    set_target_properties(${target} PROPERTIES C_VISIBILITY_PRESET hidden LIBRARY_OUTPUT_DIRECTORY "${destination}")
endfunction()
set(FEMU_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
femu_plugin(cs-femu "${FEMU_DIR}/nvme.c" "${PROJECT_BINARY_DIR}/snia-cs-api")
install(TARGETS cs-femu LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/snia-cs-api)

install(FILES include/femu-csd-kernel.h include/cs_csf.h
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/snia-cs-api/femu)
