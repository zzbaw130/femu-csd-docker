set(FEMU_DIR "${CMAKE_CURRENT_SOURCE_DIR}" PARENT_SCOPE)
set(FEMU_SOURCES)
foreach(source discovery access afdm_management csf_scheduling storage_ios management batch_scheduling)
    list(APPEND FEMU_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/femu_${source}.c")
endforeach()
foreach(source plugin_entry csf_catalog device_resources request_execution request_validation)
    list(APPEND FEMU_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/internal/femu_${source}.c")
endforeach()
set(FEMU_SOURCES "${FEMU_SOURCES}" PARENT_SCOPE)
function(femu_plugin target transport destination)
    add_library(${target} MODULE ${FEMU_SOURCES} ${transport})
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/include"
        "${PROJECT_SOURCE_DIR}/src/interface" "${FEMU_DIR}/internal" "${FEMU_DIR}/include")
    target_link_libraries(${target} PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    set_target_properties(${target} PROPERTIES C_VISIBILITY_PRESET hidden LIBRARY_OUTPUT_DIRECTORY "${destination}")
endfunction()
