cmake_minimum_required(VERSION 3.16)
project(femu_cache_test C)
find_package(Threads REQUIRED)
find_path(SNIA_INCLUDE_DIR cs.h REQUIRED)
find_library(SNIA_LIBRARY snia REQUIRED)
add_executable(test-hpcp test_hpcp.c)
target_include_directories(test-hpcp PRIVATE "${SNIA_INCLUDE_DIR}")
target_compile_options(test-hpcp PRIVATE -O2 -Wall -Wextra)
target_link_libraries(test-hpcp PRIVATE "${SNIA_LIBRARY}" Threads::Threads)
add_executable(test-cache-input test_cache_input.c)
target_include_directories(test-cache-input PRIVATE "${SNIA_INCLUDE_DIR}")
target_compile_options(test-cache-input PRIVATE -O2 -Wall -Wextra)
target_link_libraries(test-cache-input PRIVATE "${SNIA_LIBRARY}" Threads::Threads)
