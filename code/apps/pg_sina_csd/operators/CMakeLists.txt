# Both build entry points read each module's explicit EXEC declarations.
function(pg_sina_graph_operator target)
    cmake_parse_arguments(OP "RUNTIME" "" "SOURCES;EXEC" ${ARGN})
    if(OP_UNPARSED_ARGUMENTS OR NOT OP_SOURCES OR NOT OP_EXEC)
        message(FATAL_ERROR "Invalid device operator declaration for ${target}")
    endif()
    get_property(entries GLOBAL PROPERTY PG_SINA_GRAPH_SYMBOLS)
    set(exports "")
    foreach(symbol IN LISTS OP_EXEC)
        if(NOT symbol MATCHES "^[A-Za-z_][A-Za-z0-9_]*$" OR symbol IN_LIST entries)
            message(FATAL_ERROR "Invalid or duplicate EXEC symbol ${symbol}")
        endif()
        list(APPEND entries "${symbol}")
        set_property(GLOBAL APPEND_STRING PROPERTY PG_SINA_GRAPH_REGISTRY
            "    {.file=\"${target}.so\",.symbol=\"${symbol}\"},\n")
        string(APPEND exports "        ${symbol};\n")
    endforeach()
    set_property(GLOBAL PROPERTY PG_SINA_GRAPH_SYMBOLS "${entries}")
    if(NOT PG_SINA_BUILD_DEVICE_OPERATORS)
        return()
    endif()

    foreach(source IN LISTS OP_SOURCES)
        get_filename_component(source_path "${source}" REALPATH BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        string(FIND "${source_path}" "${CMAKE_CURRENT_SOURCE_DIR}/" local_source)
        if(NOT local_source EQUAL 0)
            message(FATAL_ERROR "${target} must compile sources from its own module")
        endif()
    endforeach()
    add_library(${target} SHARED ${OP_SOURCES})
    set_target_properties(${target} PROPERTIES PREFIX "" BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
    target_compile_definitions(${target} PRIVATE QP_SPLIT_RUNTIME)
    target_link_options(${target} PRIVATE "-Wl,-z,defs")
    if(OP_RUNTIME)
        target_include_directories(${target} PUBLIC
            "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../include" "${CSD_FEMU_INCLUDE}")
        target_link_libraries(${target} PUBLIC cs-workmem PRIVATE Threads::Threads)
    else()
        target_link_libraries(${target} PRIVATE pg-sina-graph-runtime)
        set_property(GLOBAL APPEND PROPERTY PG_SINA_GRAPH_TARGETS "${target}")
        # Expose the declared EXECs. Keep other definitions local to this DSO.
        set(map "${CMAKE_CURRENT_BINARY_DIR}/exports.map")
        file(GENERATE OUTPUT "${map}" CONTENT "{\n    global:\n${exports}    local: *;\n};\n")
        target_link_options(${target} PRIVATE "-Wl,--version-script=${map}" "-Wl,--no-undefined-version")
        set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS "${map}")
    endif()
    install(TARGETS ${target} LIBRARY DESTINATION .)
endfunction()

# Runtime provides the shared headers and the device primitives.
add_subdirectory(runtime)
file(GLOB modules CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "*/CMakeLists.txt")
list(SORT modules)
foreach(module IN LISTS modules)
    get_filename_component(directory "${module}" DIRECTORY)
    if(NOT directory STREQUAL "runtime")
        add_subdirectory("${directory}")
    endif()
endforeach()
if(PG_SINA_BUILD_DEVICE_OPERATORS)
    get_property(operator_targets GLOBAL PROPERTY PG_SINA_GRAPH_TARGETS)
    foreach(target IN LISTS operator_targets)
        get_target_property(libraries ${target} LINK_LIBRARIES)
        if(NOT libraries STREQUAL "pg-sina-graph-runtime")
            message(FATAL_ERROR "${target} must link only the public device runtime")
        endif()
    endforeach()
endif()
