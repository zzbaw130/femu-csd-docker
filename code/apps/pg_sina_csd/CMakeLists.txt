cmake_minimum_required(VERSION 3.16)
project(pg_sina_csd VERSION 0.2 LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 11)
find_program(PG_CONFIG pg_config)
if(NOT PG_CONFIG)
    message(FATAL_ERROR "PostgreSQL pg_config is required")
endif()
foreach(option includedir includedir-server libdir pkglibdir sharedir)
    string(REPLACE "-" "_" key "${option}")
    execute_process(COMMAND "${PG_CONFIG}" "--${option}" OUTPUT_VARIABLE PG_${key}
        OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE status)
    if(NOT status EQUAL 0)
        message(FATAL_ERROR "pg_config --${option} failed")
    endif()
endforeach()
add_library(pg_sina_csd MODULE src/extension.c src/prepare.c src/planner/aggregate.c src/planner/rewrite.c src/planner/plan.c src/executor/plan.c src/executor/aggregate.c src/executor/mapping.c src/executor/batch.c)
set_target_properties(pg_sina_csd PROPERTIES PREFIX "")
target_include_directories(pg_sina_csd PRIVATE src include "${PG_includedir_server}" "${PG_includedir}")
target_compile_definitions(pg_sina_csd PRIVATE _GNU_SOURCE)
target_compile_options(pg_sina_csd PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -fno-strict-aliasing -fwrapv)
install(TARGETS pg_sina_csd LIBRARY DESTINATION "${PG_pkglibdir}")
install(FILES pg_sina_csd.control sql/pg_sina_csd--0.1.sql sql/pg_sina_csd--0.2.sql sql/pg_sina_csd--0.1--0.2.sql DESTINATION "${PG_sharedir}/extension")
find_path(SNIA_INCLUDE cs.h)
find_library(SNIA_LIBRARY snia)
find_package(Threads REQUIRED)
find_library(PQ_LIBRARY pq HINTS "${PG_libdir}")
if(NOT SNIA_INCLUDE OR NOT SNIA_LIBRARY OR NOT PQ_LIBRARY)
    message(FATAL_ERROR "Install SNIA and PostgreSQL client development files first")
endif()
target_include_directories(pg_sina_csd PRIVATE "${SNIA_INCLUDE}")
target_link_libraries(pg_sina_csd PRIVATE "${SNIA_LIBRARY}" Threads::Threads)
add_executable(pg-sina-csd-check src/check.c)
target_include_directories(pg-sina-csd-check PRIVATE include "${SNIA_INCLUDE}" "${PG_includedir}")
target_link_libraries(pg-sina-csd-check PRIVATE "${SNIA_LIBRARY}" "${PQ_LIBRARY}")
target_compile_options(pg-sina-csd-check PRIVATE -Wall -Wextra -Werror)
if(BUILD_TESTING)
    add_executable(pg-sina-heap-layout tests/heap_layout.c)
    target_include_directories(pg-sina-heap-layout PRIVATE "${PG_includedir_server}" "${PG_includedir}")
    add_test(NAME pgSinaHeapLayout COMMAND pg-sina-heap-layout)
endif()
