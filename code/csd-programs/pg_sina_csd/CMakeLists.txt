foreach(operator scan filter aggregate sort limit)
    # Only the entry name changes; the algorithm source contains no timers.
    set(entries pg_sina_${operator})
    if(operator STREQUAL "aggregate")
        set(entries pg_sina_aggregate_sink pg_sina_aggregate_source)
    endif()
    set(renames)
    foreach(entry IN LISTS entries)
        list(APPEND renames "${entry}=${entry}_impl")
    endforeach()
    set_source_files_properties(${operator}.c PROPERTIES COMPILE_DEFINITIONS "${renames}")
    add_csd_program(pg-sina-${operator} ${operator}.c profile.c)
    string(TOUPPER "${operator}" profile_operator)
    target_compile_definitions(pg-sina-${operator} PRIVATE "PROFILE_${profile_operator}")
    target_include_directories(pg-sina-${operator} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../apps/pg_sina_csd/include")
endforeach()

    add_executable(pg-sina-query-test tests/query.c)

    target_link_libraries(pg-sina-query-test PRIVATE cs-workmem ${CMAKE_DL_LIBS})
    add_dependencies(pg-sina-query-test pg-sina-scan pg-sina-filter pg-sina-aggregate pg-sina-sort pg-sina-limit)
    set(query_test_command pg-sina-query-test $<TARGET_FILE:pg-sina-scan> $<TARGET_FILE:pg-sina-filter>
