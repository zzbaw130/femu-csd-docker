cmake_minimum_required(VERSION 3.16)
project(csd_programs LANGUAGES C CXX)
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Werror)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CSD_FEMU_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../apps/snia/src/plugins/femu/include")
# All native CSFs share one allocator and find it beside their own .so.
function(add_csd_program target)
    add_library(${target} MODULE ${ARGN})
    set_target_properties(${target} PROPERTIES PREFIX "" BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
    target_include_directories(${target} PRIVATE "${CSD_FEMU_INCLUDE}")
    target_link_libraries(${target} PRIVATE "-Wl,--push-state,--no-as-needed" cs-workmem "-Wl,--pop-state" Threads::Threads)
endfunction()
add_subdirectory(csd-workmem)
add_subdirectory(examples)
add_subdirectory(pg_sina_csd)
